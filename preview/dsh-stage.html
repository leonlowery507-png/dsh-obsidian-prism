// dsh-prism-bridge.ts — DeepSeek Harness 浏览器桥接与 512K 上下文遥测估算 (64 lines)
export interface PrismTelemetrySnapshot {
  usedTokens: number;
  maxTokens: number;
  usedFraction: number;
  ratios: [number, number, number];
  tps: number;
  ttftSeconds: number;
  streaming: boolean;
}

export interface PrismConfig {
  theme: 'obsidian' | 'warm' | 'crystal' | 'alabaster';
  accentTone: 'follow' | 'cobalt' | 'iris' | 'amber' | 'mono';
  motionLevel: 'off' | 'subtle' | 'standard' | 'ultra';
  thoughtTranslate: 'off' | 'machine' | 'ai';
  thoughtView: 'immersive' | 'bilingual';
  mascot: 'whale' | 'orca' | 'off';
}

const DEFAULT_MAX_TOKENS = 512_000;
const STORAGE_KEY = 'dsh_obsidian_prism_config';

export function estimateCjkAwareTokens(text: string): number {
  let cjkChars = 0;
  let asciiChars = 0;
  for (let i = 0; i < text.length; i += 1) {
    const code = text.charCodeAt(i);
    if (code >= 0x4e00 && code <= 0x9fff) {
      cjkChars += 1;
    } else {
      asciiChars += 1;
    }
  }
  return Math.max(1, Math.ceil(cjkChars * 1.35 + asciiChars / 3.8));
}

export function collectConversationTelemetry(root: ParentNode, streaming: boolean): PrismTelemetrySnapshot {
  const userNodes = Array.from(root.querySelectorAll('[data-role="user"]'));
  const assistantNodes = Array.from(root.querySelectorAll('[data-role="assistant"]'));
  const codeNodes = Array.from(root.querySelectorAll('pre code'));

  const systemTokens = 14_200;
  const convTokens = [...userNodes, ...assistantNodes].reduce(
    (sum, node) => sum + estimateCjkAwareTokens(node.textContent || ''),
    0,
  );
  const toolTokens = codeNodes.reduce(
    (sum, node) => sum + estimateCjkAwareTokens(node.textContent || ''),
    0,
  );

  const usedTokens = Math.min(DEFAULT_MAX_TOKENS, systemTokens + convTokens + toolTokens);
  const usedFraction = usedTokens / DEFAULT_MAX_TOKENS;
  const sum = Math.max(1, systemTokens + convTokens + toolTokens);

  return {
    usedTokens,
    maxTokens: DEFAULT_MAX_TOKENS,
    usedFraction,
    ratios: [
      (systemTokens / sum) * usedFraction,
      (convTokens / sum) * usedFraction,
      (toolTokens / sum) * usedFraction,
    ],
    tps: streaming ? 78.4 : 0,
    ttftSeconds: 0.42,
    streaming,
  };
}
